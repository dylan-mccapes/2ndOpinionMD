## SYSTEM INVARIANTS

```yaml
invariants:
  - id: INV-001
    rule: "Graph is append-only; corrections are new nodes"
    enforcement: "No UPDATE or DELETE operations on event_node or event_edge tables"
    correction_mechanism: "Create new node with status='corrected' and edge type='SUPERSEDES'"
  
  - id: INV-002
    rule: "All derived data must reference upstream nodes"
    enforcement: "Every node with source_type='derived' MUST have at least one incoming edge of type DERIVED_FROM"
    validation: "CHECK constraint on event_node table"
  
  - id: INV-003
    rule: "No embeddings without a backing graph node"
    enforcement: "Embedding tables MUST have FOREIGN KEY to event_node.event_id"
    validation: "Referential integrity enforced at database level"
  
  - id: INV-004
    rule: "No agent may reason directly from raw tables"
    enforcement: "All agent queries MUST use PatientTimelineVision API (no direct SQL to ehr.patient_timeline)"
    validation: "API gateway + code review policy"
  
  - id: INV-005
    rule: "Temporal causality must be respected"
    enforcement: "CAUSAL edges MUST point from earlier to later events (source.timestamp < target.timestamp)"
    validation: "CHECK constraint on event_edge table"
  
  - id: INV-006
    rule: "All edges must have provenance"
    enforcement: "event_edge.discovered_by MUST NOT be empty array"
    validation: "CHECK constraint: array_length(discovered_by, 1) > 0"
```

## NODE SCHEMAS

### Base Node Type

```typescript
interface PatientEventNode {
  // === IDENTITY (REQUIRED) ===
  event_id: string;              // Primary key, format: "evt_{uuid}"
  patient_id: string;            // Foreign key to patient
  version: integer;              // Node version (for append-only updates)
  
  // === CLASSIFICATION (REQUIRED) ===
  node_type: NodeType;           // See NodeType enum below
  event_subtype: string;         // Fine-grained classification
  
  // === TEMPORAL (REQUIRED) ===
  timestamp: timestamp_tz;       // Event occurrence time
  created_at: timestamp_tz;      // Node creation time (immutable)
  
  // === SOURCE (REQUIRED) ===
  source_type: SourceType;       // 'raw' | 'derived' | 'synthetic'
  source_system: string;         // EHR identifier or module name
  
  // === STATUS (REQUIRED) ===
  status: NodeStatus;            // See NodeStatus enum below
  
  // === PROVENANCE (REQUIRED) ===
  discovered_by: string[];       // Modules/agents that created or enriched this node
  
  // === DATA (OPTIONAL) ===
  structured: jsonb;             // Parsed structured data
  text: text;                    // Narrative text
  
  // === ENRICHMENT (OPTIONAL) ===
  annotations: jsonb;            // Module annotations (key = module_id)
  
  // === EMBEDDINGS (OPTIONAL) ===
  embedding_id: string | null;   // Foreign key to embedding table
  
  // === METADATA (OPTIONAL) ===
  meta: jsonb;                   // Additional metadata
}

enum NodeType {
  EVENT              = "event";              // Generic event
  MEASUREMENT        = "measurement";        // Lab, vital, test result
  MEDICATION_CHANGE  = "medication_change";  // Med start/stop/dose change
  RISK_SIGNAL        = "risk_signal";        // Flare risk, infection risk
  NOTE               = "note";               // Clinical note, report
  DECISION           = "decision";           // Treatment decision, order
  DERIVED_INSIGHT    = "derived_insight";    // EoH-generated insight (composite)
}

enum SourceType {
  RAW       = "raw";       // Ingested from EHR (ehr.patient_timeline)
  DERIVED   = "derived";   // Generated by EoH module
  SYNTHETIC = "synthetic"; // Test data or simulation
}

enum NodeStatus {
  INCLUDED   = "included";   // Active, valid node
  EXCLUDED   = "excluded";   // Filtered out (e.g., noise, artifact)
  UNCERTAIN  = "uncertain";  // Requires review
  CORRECTED  = "corrected";  // Superseded by newer node (immutable)
  DEPRECATED = "deprecated"; // No longer used
}
```

## EDGE SCHEMAS

### Base Edge Type

```typescript
interface EventRelationshipEdge {
  // === IDENTITY (REQUIRED) ===
  edge_id: string;               // Primary key, format: "edge_{uuid}"
  
  // === NODES (REQUIRED) ===
  source_event_id: string;       // Foreign key to event_node.event_id
  target_event_id: string;       // Foreign key to event_node.event_id
  
  // === CLASSIFICATION (REQUIRED) ===
  relationship_type: RelationshipType;
  
  // === STRENGTH (REQUIRED) ===
  strength: number;              // 0.0-1.0, relationship strength
  confidence: number;            // 0.0-1.0, confidence in relationship
  
  // === PROVENANCE (REQUIRED) ===
  discovered_by: string[];       // Modules/agents that created this edge (NOT NULL, length > 0)
  created_at: timestamp_tz;      // Edge creation time (immutable)
  
  // === TEMPORAL (OPTIONAL) ===
  time_delta?: interval;         // Time between source and target events
  
  // === ENRICHMENT (OPTIONAL) ===
  annotations: jsonb;            // Edge-specific annotations
  
  // === METADATA (OPTIONAL) ===
  meta: jsonb;
}

enum RelationshipType {
  // Temporal
  TEMPORAL_SEQUENCE   = "temporal_sequence";    // A before B (chronological)
  TEMPORAL_WINDOW     = "temporal_window";      // A and B in same time window
  
  // Causal
  CAUSAL_LIKELY       = "causal_likely";        // A likely caused B (strong)
  CAUSAL_POSSIBLE     = "causal_possible";      // A may have caused B (weak)
  CAUSAL_PREVENTS     = "causal_prevents";      // A prevents B
  
  // Semantic
  REFERENCES          = "references";           // A mentions/cites B
  MODIFIES            = "modifies";             // A modifies B (e.g., dose change)
  SUPERSEDES          = "supersedes";           // A corrects/replaces B
  
  // Derivation
  DERIVED_FROM        = "derived_from";         // A computed from B
  COMPOSITE           = "composite";            // A contains B (part-of)
  
  // Similarity
  CORRELATES_WITH     = "correlates_with";      // A correlates with B (statistical)
  SIMILARITY          = "similarity";           // A similar to B (features)
  
  // Conflict
  CONTRADICTS         = "contradicts";          // A conflicts with B
  
  // Planning
  CARE_PLAN_LINK      = "care_plan_link";       // A is part of care plan triggered by B
}
```

## POSTGRESQL SCHEMA (DDL)

```sql
-- =====================================================================
-- SCHEMA: ptv (PatientTimelineVision)
-- =====================================================================

CREATE SCHEMA IF NOT EXISTS ptv;

-- =====================================================================
-- TABLE: patient_vision
-- =====================================================================

CREATE TABLE ptv.patient_vision (
    patient_id          TEXT PRIMARY KEY,
    built_at            TIMESTAMPTZ NOT NULL DEFAULT now(),
    last_enriched_at    TIMESTAMPTZ,
    version             INTEGER NOT NULL DEFAULT 1,
    metadata            JSONB DEFAULT '{}'::jsonb,
    
    -- Stats
    node_count          INTEGER DEFAULT 0,
    edge_count          INTEGER DEFAULT 0,
    
    -- Provenance
    built_by            TEXT[] DEFAULT '{}',
    
    CONSTRAINT fk_patient
        FOREIGN KEY (patient_id)
        REFERENCES ehr.patient(id)
        ON DELETE CASCADE
);

-- =====================================================================
-- TABLE: event_node
-- =====================================================================

CREATE TABLE ptv.event_node (
    -- IDENTITY
    event_id            TEXT PRIMARY KEY,
    patient_id          TEXT NOT NULL,
    version             INTEGER NOT NULL DEFAULT 1,
    
    -- CLASSIFICATION
    node_type           TEXT NOT NULL,
    event_subtype       TEXT,
    
    -- TEMPORAL
    timestamp           TIMESTAMPTZ NOT NULL,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    
    -- SOURCE
    source_type         TEXT NOT NULL,
    source_system       TEXT NOT NULL,
    
    -- STATUS
    status              TEXT NOT NULL DEFAULT 'included',
    
    -- PROVENANCE (REQUIRED)
    discovered_by       TEXT[] NOT NULL,
    
    -- DATA
    structured          JSONB DEFAULT '{}'::jsonb,
    text                TEXT,
    
    -- ENRICHMENT
    annotations         JSONB DEFAULT '{}'::jsonb,
    
    -- EMBEDDINGS
    embedding_id        TEXT,
    
    -- METADATA
    meta                JSONB DEFAULT '{}'::jsonb,
    
    -- LINEAGE (for append-only corrections)
    parent_event_id     TEXT,
    superseded_by_event_id TEXT,
    
    -- CONSTRAINTS
    CONSTRAINT fk_patient_vision
        FOREIGN KEY (patient_id)
        REFERENCES ptv.patient_vision(patient_id)
        ON DELETE CASCADE,
    
    CONSTRAINT fk_parent_event
        FOREIGN KEY (parent_event_id)
        REFERENCES ptv.event_node(event_id),
    
    CONSTRAINT fk_superseded_by
        FOREIGN KEY (superseded_by_event_id)
        REFERENCES ptv.event_node(event_id),
    
    -- INV-002: Derived nodes must have upstream references (enforced via edge check)
    CONSTRAINT chk_node_type
        CHECK (node_type IN ('event', 'measurement', 'medication_change', 
                             'risk_signal', 'note', 'decision', 'derived_insight')),
    
    CONSTRAINT chk_source_type
        CHECK (source_type IN ('raw', 'derived', 'synthetic')),
    
    CONSTRAINT chk_status
        CHECK (status IN ('included', 'excluded', 'uncertain', 'corrected', 'deprecated')),
    
    -- INV-001: Append-only (enforced by trigger, not constraint)
    CONSTRAINT chk_discovered_by_not_empty
        CHECK (array_length(discovered_by, 1) > 0)
);

-- Indexes
CREATE INDEX idx_event_node_patient_ts ON ptv.event_node(patient_id, timestamp);
CREATE INDEX idx_event_node_type ON ptv.event_node(node_type);
CREATE INDEX idx_event_node_subtype ON ptv.event_node(event_subtype);
CREATE INDEX idx_event_node_status ON ptv.event_node(status);
CREATE INDEX idx_event_node_source_type ON ptv.event_node(source_type);
CREATE INDEX idx_event_node_structured_gin ON ptv.event_node USING GIN(structured);
CREATE INDEX idx_event_node_annotations_gin ON ptv.event_node USING GIN(annotations);

-- =====================================================================
-- TABLE: event_edge
-- =====================================================================

CREATE TABLE ptv.event_edge (
    -- IDENTITY
    edge_id             TEXT PRIMARY KEY,
    
    -- NODES
    source_event_id     TEXT NOT NULL,
    target_event_id     TEXT NOT NULL,
    
    -- CLASSIFICATION
    relationship_type   TEXT NOT NULL,
    
    -- STRENGTH
    strength            FLOAT NOT NULL,
    confidence          FLOAT NOT NULL,
    
    -- PROVENANCE (REQUIRED)
    discovered_by       TEXT[] NOT NULL,
    created_at          TIMESTAMPTZ NOT NULL DEFAULT now(),
    
    -- TEMPORAL
    time_delta          INTERVAL,
    
    -- ENRICHMENT
    annotations         JSONB DEFAULT '{}'::jsonb,
    
    -- METADATA
    meta                JSONB DEFAULT '{}'::jsonb,
    
    -- CONSTRAINTS
    CONSTRAINT fk_source_event
        FOREIGN KEY (source_event_id)
        REFERENCES ptv.event_node(event_id)
        ON DELETE CASCADE,
    
    CONSTRAINT fk_target_event
        FOREIGN KEY (target_event_id)
        REFERENCES ptv.event_node(event_id)
        ON DELETE CASCADE,
    
    CONSTRAINT chk_relationship_type
        CHECK (relationship_type IN (
            'temporal_sequence', 'temporal_window',
            'causal_likely', 'causal_possible', 'causal_prevents',
            'references', 'modifies', 'supersedes',
            'derived_from', 'composite',
            'correlates_with', 'similarity',
            'contradicts',
            'care_plan_link'
        )),
    
    CONSTRAINT chk_strength_range
        CHECK (strength BETWEEN 0.0 AND 1.0),
    
    CONSTRAINT chk_confidence_range
        CHECK (confidence BETWEEN 0.0 AND 1.0),
    
    -- INV-006: All edges must have provenance
    CONSTRAINT chk_discovered_by_not_empty
        CHECK (array_length(discovered_by, 1) > 0),
    
    -- Prevent self-loops (optional, may be relaxed for specific edge types)
    CONSTRAINT chk_no_self_loops
        CHECK (source_event_id != target_event_id)
);

-- Indexes
CREATE INDEX idx_event_edge_source ON ptv.event_edge(source_event_id);
CREATE INDEX idx_event_edge_target ON ptv.event_edge(target_event_id);
CREATE INDEX idx_event_edge_type ON ptv.event_edge(relationship_type);
CREATE INDEX idx_event_edge_strength ON ptv.event_edge(strength);
CREATE INDEX idx_event_edge_annotations_gin ON ptv.event_edge USING GIN(annotations);

-- Composite indexes for traversal
CREATE INDEX idx_event_edge_source_type ON ptv.event_edge(source_event_id, relationship_type);
CREATE INDEX idx_event_edge_target_type ON ptv.event_edge(target_event_id, relationship_type);

-- =====================================================================
-- TABLE: node_embeddings (INV-003: No embeddings without backing node)
-- =====================================================================

CREATE TABLE ptv.node_embeddings (

-- =====================================================================
-- TRIGGERS (INV-001: Append-only enforcement)
-- =====================================================================

CREATE OR REPLACE FUNCTION ptv.prevent_node_updates()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'UPDATE' THEN
        RAISE EXCEPTION 'INV-001 violation: Updates not allowed on event_node. Create new node with parent_event_id instead.';
    END IF;
    IF TG_OP = 'DELETE' THEN
        RAISE EXCEPTION 'INV-001 violation: Deletes not allowed on event_node. Update status to deprecated instead.';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_prevent_node_updates
    BEFORE UPDATE OR DELETE ON ptv.event_node
    FOR EACH ROW
    EXECUTE FUNCTION ptv.prevent_node_updates();

CREATE OR REPLACE FUNCTION ptv.prevent_edge_updates()
RETURNS TRIGGER AS $$
BEGIN
    IF TG_OP = 'UPDATE' THEN
        RAISE EXCEPTION 'INV-001 violation: Updates not allowed on event_edge. Create new edge instead.';
    END IF;
    IF TG_OP = 'DELETE' THEN
        RAISE EXCEPTION 'INV-001 violation: Deletes not allowed on event_edge.';
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_prevent_edge_updates
    BEFORE UPDATE OR DELETE ON ptv.event_edge
    FOR EACH ROW
    EXECUTE FUNCTION ptv.prevent_edge_updates();

-- =====================================================================
-- VALIDATION FUNCTIONS (INV-002: Derived nodes must have upstream refs)
-- =====================================================================

CREATE OR REPLACE FUNCTION ptv.validate_derived_node_has_upstream()
RETURNS TRIGGER AS $$
DECLARE
    upstream_count INTEGER;
BEGIN
    IF NEW.source_type = 'derived' AND NEW.node_type = 'derived_insight' THEN
        -- Check if at least one DERIVED_FROM edge exists
        SELECT COUNT(*) INTO upstream_count
        FROM ptv.event_edge
        WHERE target_event_id = NEW.event_id
          AND relationship_type = 'derived_from';
        
        IF upstream_count = 0 THEN
            RAISE EXCEPTION 'INV-002 violation: Derived node % must have at least one DERIVED_FROM edge', NEW.event_id;
        END IF;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- NOTE: This trigger fires AFTER INSERT to allow edge creation first
-- Consider enforcing at application level or via deferred constraint

-- =====================================================================
-- VALIDATION FUNCTIONS (INV-005: Temporal causality)
-- =====================================================================

CREATE OR REPLACE FUNCTION ptv.validate_causal_temporal_ordering()
RETURNS TRIGGER AS $$
DECLARE
    source_ts TIMESTAMPTZ;
    target_ts TIMESTAMPTZ;
BEGIN
    IF NEW.relationship_type IN ('causal_likely', 'causal_possible', 'causal_prevents') THEN
        SELECT timestamp INTO source_ts
        FROM ptv.event_node
        WHERE event_id = NEW.source_event_id;
        
        SELECT timestamp INTO target_ts
        FROM ptv.event_node
        WHERE event_id = NEW.target_event_id;
        
        IF source_ts >= target_ts THEN
            RAISE EXCEPTION 'INV-005 violation: Causal edge from % to % violates temporal ordering (source: %, target: %)',
                NEW.source_event_id, NEW.target_event_id, source_ts, target_ts;
        END IF;
        
        -- Auto-compute time_delta
        NEW.time_delta := target_ts - source_ts;
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trigger_validate_causal_temporal_ordering
    BEFORE INSERT ON ptv.event_edge
    FOR EACH ROW
    EXECUTE FUNCTION ptv.validate_causal_temporal_ordering();

-- =====================================================================

CREATE OR REPLACE VIEW ptv.v_active_nodes AS
SELECT *
FROM ptv.event_node
WHERE status = 'included'
  AND superseded_by_event_id IS NULL;

CREATE OR REPLACE VIEW ptv.v_causal_graph AS
SELECT
    e.edge_id,
    e.source_event_id,
    e.target_event_id,
    e.relationship_type,
    e.strength,
    e.confidence,
    e.time_delta,
    sn.timestamp AS source_timestamp,
    sn.node_type AS source_type,
    sn.event_subtype AS source_subtype,
    tn.timestamp AS target_timestamp,
    tn.node_type AS target_type,
    tn.event_subtype AS target_subtype
FROM ptv.event_edge e
JOIN ptv.event_node sn ON e.source_event_id = sn.event_id
JOIN ptv.event_node tn ON e.target_event_id = tn.event_id
WHERE e.relationship_type IN ('causal_likely', 'causal_possible', 'causal_prevents')
  AND sn.status = 'included'
  AND tn.status = 'included';
```

```sql
-- =====================================================================
-- MIGRATION: ehr.patient_timeline → ptv.event_node
-- =====================================================================

CREATE OR REPLACE FUNCTION ptv.ingest_from_ehr_timeline(p_patient_id TEXT)
RETURNS TABLE(events_ingested INTEGER, edges_created INTEGER) AS $$
DECLARE
    event_count INTEGER := 0;
    edge_count INTEGER := 0;
    prev_event_id TEXT := NULL;
    curr_event_id TEXT;
    curr_timestamp TIMESTAMPTZ;
    prev_timestamp TIMESTAMPTZ;
BEGIN
    -- Create patient vision if not exists
    INSERT INTO ptv.patient_vision (patient_id, built_by)
    VALUES (p_patient_id, ARRAY['ehr_timeline_ingestion'])
    ON CONFLICT (patient_id) DO NOTHING;
    
    -- Ingest events from ehr.patient_timeline
    FOR curr_event_id, curr_timestamp IN
        SELECT 
            'evt_' || id::text AS event_id,
            ts AS timestamp
        FROM ehr.patient_timeline
        WHERE patient_id = p_patient_id
        ORDER BY ts ASC
    LOOP
        -- Insert node
        INSERT INTO ptv.event_node (
            event_id,
            patient_id,
            node_type,
            event_subtype,
            timestamp,
            source_type,
            source_system,
            discovered_by,
            structured,
            text,
            meta
        )
        SELECT
            curr_event_id,
            patient_id,
            CASE
                WHEN event_type = 'lab' THEN 'measurement'
                WHEN event_type = 'med' THEN 'medication_change'
                WHEN event_type = 'symptom' THEN 'measurement'
                WHEN event_type = 'note' THEN 'note'
                ELSE 'event'
            END::text,
            event_subtype,
            ts,
            'raw',
            source,
            ARRAY['ehr_timeline_ingestion'],
            structured,
            text,
            meta
        FROM ehr.patient_timeline
        WHERE id::text = substring(curr_event_id from 5);
        
        event_count := event_count + 1;
        
        -- Create TEMPORAL_SEQUENCE edge to previous event
        IF prev_event_id IS NOT NULL THEN
            INSERT INTO ptv.event_edge (
                edge_id,
                source_event_id,
                target_event_id,
                relationship_type,
                strength,
                confidence,
                discovered_by,
                time_delta
            ) VALUES (
                'edge_' || gen_random_uuid()::text,
                prev_event_id,
                curr_event_id,
                'temporal_sequence',
                1.0,
                1.0,
                ARRAY['ehr_timeline_ingestion'],
                curr_timestamp - prev_timestamp
            );
            edge_count := edge_count + 1;
        END IF;
        
        prev_event_id := curr_event_id;
        prev_timestamp := curr_timestamp;
    END LOOP;
    
    -- Update patient vision stats
    UPDATE ptv.patient_vision
    SET 
        node_count = event_count,
        edge_count = edge_count,
        last_enriched_at = now()
    WHERE patient_id = p_patient_id;
    
    RETURN QUERY SELECT event_count, edge_count;
END;
$$ LANGUAGE plpgsql;
```
